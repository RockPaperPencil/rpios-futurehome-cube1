# This config is intended to be included from config.txt!
# It is meant to be included in [blocks] for the SODIMM compute modules

[gpio41=1]
# CUBE-1v0 SECTION. Matched/triggered by gpio41 reading high.
# The case LED initially shining RED when booting means that a cube-1v0 
# carrier board was detected.
gpio=11=op,dh

# The cube-1v0 Z-Wave radio UART interface is traced out to GPIOs 40/41. The
# only available UART on those GPIOs is the mini-UART. The Z-Wave UART is
# therefore at /dev/ttyS0 without any other option.
dtoverlay=uart1,txd1_pin=40,rxd1_pin=41

# The cube-1v0 EnOcean radio UART interface is traced out to GPIOs 14/15. The 
# PL011 UART is used as it simply is not assignable to the Z-Wave module.
# EnOcean UART is therefore at /dev/ttyAMA0.
dtoverlay=uart0,txd0_pin=14,rxd0_pin=15,pin_func=4

include config-fhcube-common.txt

[gpio33=1]
# CUBE-1v1 SECTION. Matched/triggered by gpio33 reading high.
# The case LED initially shining GREEN when booting means that a cube-1v1 
# carrier board was detected.
gpio=12=op,dh

# The cube-1v1 Z-Wave radio UART interface is traced out to GPIOs 32/33. These
# GPIOs can have the PL011 UART mapped to them, so we naturally do that here.
# Z-Wave in the cube-1v1 is therefore available at /dev/ttyAMA0.
dtoverlay=uart0,txd0_pin=32,rxd0_pin=33,pin_func=7

# Disabled by default; /dev/ttyS0 at DBG header pins 4/5
# dtoverlay=uart1,txd1_pin=14,rxd1_pin=15

include config-fhcube-common.txt
